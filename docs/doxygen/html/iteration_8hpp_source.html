147template <typename T>
148class IteratorFacade : public std::input_iterator_tag {
149 public:
150 using value_type = Handle<T>;
151
153 using self = IteratorFacade<value_type>;
154 using iterator = Iterator<Raw<value_type>>;
155
158 explicit IteratorFacade(iterator it, bool at_end = false)
159 : it_(it), current_value_(std::nullopt) {
160 if (!at_end) {
161 UncheckedAdvance();
162 }
163 }
164
167 self& operator++() {
168 Advance();
169 return *this;
170 }
171
172 self operator++(int) {
173 auto res = *this;
174 ++(*this);
175
176 return res;
177 }
178
179 value_type operator*() {
180 if (!current_value_) {
181 throw ValueError(
182 "Mamba-internal error: Cannot dereference an iterator that has "
183 "already thrown StopIterator.");
184 }
185
186 return *current_value_;
187 }
188
189 bool operator==(const self& other) const {
190 return !!current_value_ == !!other.current_value_ && it_ == other.it_;
191 }
192
193 bool operator!=(const self& other) const { return !(*this == other); }
194
195 self begin() const { return *this; }
196
197 self end() const {
198 auto res = *this;
199 res.current_value_.reset();
200
201 return res;
202 }
203
204 private:
205 void UncheckedAdvance() {
206 try {
207 current_value_ = it_.__Next__();
208 } catch (StopIteration) {
209 current_value_.reset();
210 }
211 }
212
213 void Advance() {
214 if (!current_value_) {
215 // No-op
216 return;
217 }
218
219 UncheckedAdvance();
220 }
221
222 // C++ only facility, no need to create std::shared_ptr<Data> over these
223 // members.
224 iterator it_;
225 std::optional<value_type> current_value_;
226};
