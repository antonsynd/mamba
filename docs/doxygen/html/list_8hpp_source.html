631 std::optional<std::pair<size_t, size_t>> TryGetNormalizedSliceIndices(
632 details::Int start,
633 details::Int end,
634 details::Int step) const {
635 // Zero step is invalid

706 void ReplaceSliceSingleStep(const self& other,
707 SliceParams<iterator> slice_params) {
708 const auto start_it = std::move(slice_params.start_it);
709 const auto start = slice_params.start;
710 const auto end = slice_params.end;
711
712 const auto num_old_elems = end - start;
713 const auto num_new_elems = other.data_.v_.size();
714
715 if (num_old_elems < num_new_elems) {
716 ReplaceSliceSingleStepExpanding(other, start, num_old_elems,
717 num_new_elems);
718 } else if (num_old_elems > num_new_elems) {
719 ReplaceSliceSingleStepReducing(other, start_it, num_old_elems,
720 num_new_elems);
721 } else {
722 // Trivial case, replace 1-to-1
723 std::copy(other.data_.v_.begin(), other.data_.v_.end(), start_it);
724 }
725 }
726
727 void ReplaceSliceSingleStepExpanding(const self& other,
728 size_t start,
729 size_t num_old_elems,
730 size_t num_new_elems) {
731 const auto num_extra_elems = num_new_elems - num_old_elems;
732 data_.v_.resize(data_.v_.size() + num_extra_elems);
733
734 // Recalculate the start iterator in case the resize() call
735 // invalidated it
736 const auto start_it = data_.v_.begin() + start;
737
738 // Shift elements from the starting position to make room for the
739 // incoming ones
740 std::shift_right(start_it, data_.v_.end(), num_extra_elems);
741
742 // Copy into the desired range
743 std::copy(other.data_.v_.begin(), other.data_.v_.end(), start_it);
744 }
745
746 void ReplaceSliceSingleStepReducing(const self& other,
747 iterator start_it,
748 size_t num_old_elems,
749 size_t num_new_elems) {
750 const auto num_elems_to_remove = num_old_elems - num_new_elems;
751
752 // Copy into desired range
753 start_it =
754 std::copy(other.data_.v_.begin(), other.data_.v_.end(), start_it);
755 const auto end_it = start_it + num_elems_to_remove;
756
757 // Erase leftover elements
758 data_.v_.erase(start_it, end_it);
759 }

761 void ReplaceSliceMultiStep(const self& other,
762 SliceParams<iterator> slice_params) {
763 const auto [start, end, step, start_it, end_it] = std::move(slice_params);
764 const auto num_old_elems = GetNumberOfElementsInSlice(start, end, step);
765
766 if (other.data_.v_.size() != num_old_elems) {
767 throw ValueError(
768 "ValueError: attempt to assign sequence of size {} to extended "
769 "slice "
770 "of size {}");
771 }
772
773 size_t idx = 0;
774 auto other_it = other.data_.v_.begin();
775 const auto other_end = other.data_.v_.end();
776
777 std::for_each(start_it, end_it,
778 [&idx, step, &other_it, &other_end](value_type& elem) {
779 if (other_it == other_end) {
780 // Shouldn't happen, but here as a fail-safe
781 return;
782 }
783
784 if (idx % step == 0) {
785 elem = *other_it;
786 ++other_it;
787 }
788
789 ++idx;
790 });
791 }
